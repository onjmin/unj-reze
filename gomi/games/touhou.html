// 開幕ボス移動
moveBoss(-200, -900, 60)
setBossLife(8000)
name = "氷符「ブルーフロストオーロラ」"
timeLimit = 45
// ループ弾幕
for t in range(0, 900, 6)
base = t*4
for i in range(0, 12, 1)
ang = base + i*30
id = shotA(0, 5, ang, 3.0, 0)
setColor(id, 5)
setScale(id, 0.9)
end for
// プレイヤー狙いサブ
a2 = getPlayerAngle()
id2 = shot2A(0, 3, a2 + rand(-10,10), 2.2, 0.05, 90)
setColor(id2, 4)
end for
